243 // INPLACE_VECTOR
245 template <typename T, size_t N, typename FCT>
246 static auto iterate(bool transient, int depth, const inplace_vector<T, N>& t, FCT fct,
247 void* context = nullptr) {
248 DEBUG_SERIALIZATION(DEBUG_FMT, depth, "iterate", "inplace_vector", (uint32_t)(N * sizeof(T)));
249
250 serial_size_t n = t.size();
251 iterate(true, depth, n, fct, context);
252
253 if (n > 0) {
254 fct(transient, depth, (void*)t.data(), sizeof(T) * n, round(sizeof(T) * n));
255 } else {
256 // in case the vector is empty, we still provide some data in order not to have an nullptr.
257 auto [dummy, DN] = getDummyBuffer();
258 fct(transient, depth, dummy, DN, round(DN));
259 }
260 }

262 template <typename T, size_t N>
263 static auto restore(iter_t& it, typename ARCH::template inplace_vector<T, N>& result) {
264 serial_size_t n = 0;
265 restore(it, n);
266
267 DEBUG_SERIALIZATION(DEBUG_FMT, 0, "restore", "inplace_vector", (uint32_t)n);
268
269 // We resize the vector.
270 result.reserve(round(n * sizeof(T)));
271 result.resize(n);
272
273 it.memcpy((char*)result.data(), round(n * sizeof(T)));
274 it.advance(round(n * sizeof(T)));
275
276 return true;
277 }

554 DEBUG_SERIALIZATION ("[size] ++++++++++++++++++\n");
555
556 size_t idx = 0;
557 pair<size_t,size_t> sizes {0,0}; // transient,permanent
558 iterate (false, 0, x, [&] (bool transient, int depth, void* ptr, size_t size, size_t roundedSize)

560 callback (transient, size, roundedSize);
561 sizes.first += transient ? roundedSize : 0 ;
562 sizes.second += not transient ? roundedSize : 0 ;
563 idx++;
564 });
565 DEBUG_SERIALIZATION ("[size] ------------------ %d,%d \n", uint32_t(sizes.first),uint32_t(sizes.first) );

575 template<typename T>
576 static buffer_t to (const T& x)
577 {
578 // First iteration: get the total size.
579 // Note: we also could use only one iteration and resize the vector size at each iteration.
580 // -> however, computing first the full vector size allows to have only one vector memory allocation.
581 auto totalSize = size (x);
582
583 DEBUG_SERIALIZATION ("[to ] ++++++++++++++++++\n");
584
585 // Second iteration: fill the buffer
586 buffer_t buffer (totalSize.first + totalSize.second);
587 auto loop = buffer.data();
588 iterate (false, 0, x, [&] (bool transient, int depth, void* ptr, size_t size, size_t roundedSize)
589 {
590 DEBUG_SERIALIZATION ("[to ] ------------------------------------------> dist: %ld\n", loop - buffer.data());
591 // We can do a memcpy here since the data pointed by 'ptr' is supposed
592 // to be available at this time in memory. We are also sure about the size
593 // available in 'buffer' since its size has been previously computed.
594 memcpy (loop, (char*)ptr, size);
595 loop += roundedSize;
596 });
597
598 return buffer;
599 }

601 template<typename INFO, typename TRANSFO, typename CALLBACK, typename ...ARGS>
602 static size_t tuple_to_buffer (const tuple<ARGS...>& x, buffer_t& buffer, INFO info, TRANSFO transfo, CALLBACK cbk)
603 {
604 // The serialized buffer is built by serializing each item of the provided tuple.
605 // A previous version computed first the size of the buffer in order to allocate
606 // the correct size. Now, we prefer to build the buffer dynamically and update the
607 // buffer size when needed. Actually, we provided a initial 'defaultSize' size that
608 // could be enough to cope with several small sizeof for the tuple items.
609 [[maybe_unused]] size_t defaultSize = 256;
610
611 size_t idxArg = 0;
612 size_t offsetLocal = 0;
613 size_t offsetGlobal = 0;
614 size_t offsetTransient = 0;
615
616 // We need to compute the buffer size first since we will provide addresses to the 'cbk' and not offsets
617 // from a base address (that could change over time if the buffer needs to be resized).
618 // The buffer size should include only the size of transient objects. For instance, arguments that are vectors
619 // are supposed to exist during the whole serialization process, so their data don't need to be copied in the
620 // buffer. On the other hand, the size of a vector requires the allocation of a temporary object that holds the
621 // vector size; in such a case, the object is transient and needs to be copied in the buffer.
622
623 size_t transientBufferSize = 0;
624
625 size_t iarg=0;
626 for_each_in_tuple (x, [&] (auto&& itemTuple)
627 {
628 // see https://stackoverflow.com/questions/46114214/lambda-implicit-capture-fails-with-variable-declared-from-structured-binding
629 size_t level, isSplit, nbUnits;
630 std::tie(level, isSplit, nbUnits) = info (iarg, std::forward<decltype(itemTuple)>(itemTuple));
631
632 // We get the size of the current argument.
633 size (itemTuple, [&] (bool transient, size_t size, size_t roundedSize)
634 {
635 if (transient) { transientBufferSize += roundedSize * nbUnits; }
636 });
637
638 iarg++;
639 });
640
641 // We reserve only for transient data
642 buffer.resize (transientBufferSize);
643
644 // We need to know the number of parts iterated for each argument.
645 // These numbers will be known for sure after the parsing of the first DPU and then can
646 // be reused for other arguments as an offet.
647 size_t nbPartsPerArgOffset[32]; for (auto& x : nbPartsPerArgOffset) { x=0; }
648
649 for_each_in_tuple (x, [&] (auto&& itemTuple)
650 {
651 DEBUG_SERIALIZATION ("[tuple_to_buffer ] ------------------------------------------> idxArg:%ld local: %ld global: %ld\n", idxArg, offsetLocal, offsetGlobal);
652
653 // For each item of the tuple, we apply a transformation.
654 // Such a transformation may for instance split the item in several parts (one per DPU for instance).
655 size_t idxDPU = 0;
656
657 auto fct = [&] (bool status, auto&& item)
658 {
659 DEBUG_SERIALIZATION ("[tuple_to_buffer ] status: %d\n", status);
660
661 size_t idxPart = 0; // for instance, a vector is serialized in two parts.
662
663 // We serialize the item only if the status allows to do so.
664 if (status)
665 {
666 offsetLocal = 0;
667
668 // lifecycle object : true-> transient false->permanent
669 iterate (false, 0, item, [&] (bool isTransient, int depth, void* ptr, size_t size, size_t roundedSize)
670 {
671 // Here, we are in the following loop:
672 // foreach idxArg
673 // foreach idxDPU
674 // foreach idxPart
675 //
676 // So we have to compute the position of the current block (in the scatter/gather DPU point of view)
677 // for the current DPU. Example:
678 //
679 // DPU0: ARG0 ARG1 ARG2
680 // PART0 PART1 PART0 PART0 PART1
681 // DPU1: ARG0 ARG1 ARG2
682 // PART0 PART1 PART0 PART0 PART1
683
684 size_t idxBlock = nbPartsPerArgOffset[idxArg] + idxPart;
685
686 uint8_t* actualPtr = (uint8_t*) ptr;
687
688 if (isTransient)
689 {
690 // We may have to resize the buffer if needed. Note that we add a few extra 'defaultSize' bytes
691
692 // We can do a memcpy here since the data pointed by 'ptr' is supposed
693 // to be available at this time in memory. We are also sure about the size
694 // available in 'buffer' since its size has been previously updated if needed.
695 memcpy (buffer.data() + offsetTransient, (char*)ptr, size);
696
697 actualPtr = (uint8_t*) buffer.data() + offsetTransient;
698
699 offsetTransient += roundedSize;
700 }
701
702 VERBOSE_SERIALIZATION ("[tuple_to_buffer] transient: %d sts: %d idxDPU: %3ld idxArg: %3ld idxPart: %2ld idxBlock: %3ld nbPtsOff: %ld offsets: [%10ld,%7ld,%7ld] #buf: %10d size: %8d ptr: [%p,%p] \n",
703 isTransient, status, idxDPU, idxArg, idxPart, idxBlock, nbPartsPerArgOffset[idxArg],
704 offsetGlobal, offsetLocal,offsetTransient,
705 (uint32_t)buffer.size(),
706 (uint32_t)size, ptr, actualPtr
707 );
708
709 // We call the callback in order to provide some information during the serialization process.
710 // NOTE: we could have used a std::vector in order to gather this information and then provide it
711 // to the caller. However, the serialize capacity we implement here can be used either by the multicore
712 // arch and the UPMEM arch, so we prefer here make no use of explicit std class but nevertheless provide
713 // some information through a callback.
714
715 // We invoke the callback for the current part.
716 cbk (idxDPU, idxArg, idxBlock, actualPtr, roundedSize);
717
718 offsetLocal += roundedSize;
719 offsetGlobal += roundedSize;
720
721 // We increase the parts number for the current argument to be serialized.
722 idxPart++;
723 });
724 }
725
726 else // not status
727 {
728 //printf ("==> iterate sts: %d idxDPU: %3ld idxArg: %3ld \n", status, idxDPU, idxArg);
729
730 // We have to reuse the same scheme as the previous DPU.
731 // By convention, we set the pointer to null in order to inform the caller
732
733 for (size_t idxBlock = nbPartsPerArgOffset[idxArg]; idxBlock<nbPartsPerArgOffset[idxArg+1]; idxBlock++)
734 {
735 cbk (idxDPU, idxArg, idxBlock, nullptr, 0);
736 }
737 }
738
739 // We remember the number of iterated parts for the current argument.
740 // We can do that on the first DPU for instance.
741 if (idxDPU==0)
742 {
743 // We need to compute the sum of the previous values.
744 nbPartsPerArgOffset [idxArg+1] = idxPart + nbPartsPerArgOffset [idxArg];
745 }
746
747 idxDPU++;
748 };
749
750 // We get some information about the current item of the tuple.
751 // In particular, we are interested to know whether it has to be split or not.
752 auto [level, isSplit, nbUnits] = info (idxArg, std::forward<decltype(itemTuple)>(itemTuple));
753
754 DEBUG_SERIALIZATION ("[tuple_to_buffer ] level: %d isSplit: %d nbUnits: %ld \n", level, isSplit, nbUnits);
755
756 // According to the split status, we don't have the same action.
757 if (not isSplit)
758 {
759 // No split -> we can use the same item.
760 // For instance with Upmem arch, it means that the same information will be broadcasted to all DPUs.
761 for (size_t i=0; i<nbUnits; i++) { fct (i==0, std::forward<decltype(itemTuple)>(itemTuple)); }
762 }
763 else
764 {
765 // We need a static check here -> avoid to call this code for non 'splitable' types (otherwise compil error).
766 // This is actually not very satisfying and should be improved.
767 if constexpr(is_splitable_v<decltype(itemTuple)>)
768 {
769 bool status = true;
770
771 // Split -> several items will be created (one per split part)
772 for (auto&& itemRef : transfo(idxArg, std::forward<decltype(itemTuple)>(itemTuple)))
773 {
774 fct (status, std::forward<decltype(itemRef)>(itemRef));
775 }
776 }
777 }
778
779 idxArg++;
780 });
781
782 size_t idx = 0;
783 for (__attribute__((unused)) unsigned char c : buffer)
784 {
785 if (idx % 32 == 0) { VERBOSE_SERIALIZATION ("\nhost2dpu 0x%04lx> ", idx); }
786 VERBOSE_SERIALIZATION ("%4d ", c);
787 if (idx % 8 == 7) { VERBOSE_SERIALIZATION ("|"); }
788 idx++;
789 }
790 VERBOSE_SERIALIZATION ("\n");
791
792 return offsetGlobal;
793 }
