25struct Cache

27 //static_assert ((NB_ITEMS & (NB_ITEMS - 1)) == 0); // supposed to be a power of 2

29 using address_t = typename ALLOCATOR::address_t;

44 auto read (address_t a, address_t begin=0, address_t end=0) {
45 address_= a; boundaries_[0]=begin; boundaries_[1]=end;
46 ALLOCATOR::read ((void*)address_, (void*)datablock_, sizeof(datablock_));
47 dump("READ ");
48 }
49
50 template<int K>
51 auto readNext () { read (address()+ K*SIZEOF_DATABLOCK); }
52
53 template<int K, auto cond=[] (ssize_type){ return true; }>
54 auto updateIfFull (ssize_type idx) {
55 if (isFull(idx) and cond(idx)) { readNext<K>(); }
56 }
57
58 auto update () {
59 dump("UPDATE");
60 if (boundaries_[1]==0) {
61 return ALLOCATOR::writeAt ((void*)address_, (void*)datablock_, sizeof(datablock_));
62 }
63 else {
64 auto a0 = std::max (address_, boundaries_[0]);
65 auto a1 = std::min (address_+sizeof(datablock_), boundaries_[1]);
66 return ALLOCATOR::writeAt (
67 (void*) a0,
68 (void*)datablock_,
69 (a1-a0)
70 );
71 }
72 }
73
74 auto write () { dump("WRITE"); address_= ALLOCATOR::write ( (void*)datablock_, sizeof(datablock_)); return address_; }
75
76 auto acquire (size_type nbitems=0) {
77 dump("ACQUIRE");
78 // By default, we get a size equal to the cache size.
79 size_type size = (1+nbitems/NB_ITEMS)*SIZEOF_DATABLOCK;
80 address_= ALLOCATOR::get(size);
81 for (int i=0; i<NB_ITEMS; i++) { datablock_[i] = {}; }
82 return address_;
83 }
84
85 T& operator [] (size_type idx) const { return datablock_[idx & MASK]; }
86
87 bool isFull (size_type idx) const { return (idx & MASK) == MASK; }
88
89 auto address() const { return address_; }
90
91 auto dump (const char* msg)
92 {
93 DEBUG_VECTOR ("[%s this: 0x%lx address: 0x%04lx: ", msg, uint64_t(this), uint64_t(address_));
94 for (int i=0; i<NB_ITEMS; i++) { DEBUG_VECTOR ("%3ld ", uint64_t(datablock_[i])); }
95 DEBUG_VECTOR ("]\n");
96 }

98 static size_type getBlockIdx (size_type idx) {
99 // We might optimize the computation in case the number of items is a power of two.
100 if constexpr ((NB_ITEMS & (NB_ITEMS - 1)) == 0) {
101 return idx >> bpl::Log2<NB_ITEMS>::value;
102 }
103 else {
104 return idx / NB_ITEMS;
105 }
106 }

108 void reset_state() { address_ = 0; }

112 ARCH_ALIGN address_t address_ = 0;
113 ARCH_ALIGN address_t boundaries_[2] = {0,0};

115 static constexpr int SIZEOF_DATABLOCK = sizeof(datablock_);
116};

288 auto data() const {
289 // We need some trick to access the raw data at a given offset according to alignment constraints.
290 // (might need a read in MRAM in UPMEM case)
291 struct RawData {
292 vector_view const& ref_;
293 T* ptr = nullptr;
294 auto operator+ (size_t idx) const {
295 auto offset = ref_.getFillAddress() + idx;
296 uint8_t delta = offset % 8;
297 ref_.cache().read (offset - delta);
298 return RawData { ref_, ref_.cache().datablock_ + delta };
299 }
300 operator T*() const { return ptr; }
301 };
302 return RawData { *this, cache().datablock_ };
303 }
304
305 value_type const& operator[] (size_type idx) const
306 {
307 // We may need synchronization. Note that 'empty' impl won't do any synchronization.
308 mutex_.lock();
309
310 // we get the block index where the required item should be.
311 auto block = cache_t::getBlockIdx(idx);
312
313 if ((size_type)block != currentBlock_)
314 {
315 // we remember the current block.
316 currentBlock_ = block;
317
318 // we read the new block from the memory
319 this->cache().read (this->getFillAddress() + currentBlock_*SIZEOF_DATABLOCK);
320 }
321
322 mutex_.unlock();
323
324 return this->cache() [idx];
325 }

327 value_type at (size_type idx) const {
328 value_type result = {};
329 ALLOCATOR::read ((T*)this->getFillAddress() + idx, (void*)&result, sizeof(result));
330 return result;

548 NOINLINE void push_back (const T& item)
549 {
550 VERBOSE_VECTOR ("[vector::push_back] idx before: %d cachefull: %d idxCache_: %ld previousBlockIdx_[idxCache_]: %ld \n", this->datablockIdx_,
551 this->cache(idxCache_).isFull(this->datablockIdx_),
552 uint64_t(idxCache_),
553 uint64_t(previousBlockIdx_[idxCache_])
554 );
555
556 // We check that we the current data block is not full
557 if (this->cache(idxCache_).isFull(this->datablockIdx_))
558 {
559 // We flush the vector (cache + memtree).
560 flush_block (true);
561
562 // We also remember the block id for a possible call to the operator []
563 previousBlockIdx_[idxCache_] = cache_t::getBlockIdx(this->datablockIdx_+1);
564 }
565
566 // we increase the index
567 this->datablockIdx_++;
568
569 // we add the provided item in the datablock
570 this->cache(idxCache_)[this->datablockIdx_] = item;
571
572 this->dirty_ = true;
573 }

583 void reset_state()
584 {
585 vector_view<T,ALLOCATOR,MUTEX,MEMORY_SIZE_LOG2,CACHE_NB_LOG2,SHARED_ITER_CACHE>::reset_state();

596 mutable size_t idxCache_ = 0;
597
598 value_type& operator[] (size_type idx) const
599 {
600 VERBOSE_VECTOR ("vector::operator[%4ld] CACHE_MEMORY_SIZE_LOG2: %ld CACHE_NB: %ld CACHE_NB_ITEMS: %ld this: 0x%lx \n",
601 (uint64_t)idx,
602 (uint64_t)parent_t::CACHE_MEMORY_SIZE_LOG2,
603 (uint64_t)parent_t::CACHE_NB,
604 (uint64_t)parent_t::CACHE_NB_ITEMS,
605 (uint64_t)this
606 );
607
608 // We look for a cache that already holds the required information
609 size_t look=0;
610 size_type blockIdx = cache_t::getBlockIdx(idx);
611 for (look=0; look<CACHE_NB; look++)
612 {
613 if (blockIdx == previousBlockIdx_[look]) { break; }
614 }
615
616 if (look==CACHE_NB)
617 {
618 // We need to get the address where the datablock needs to be retrieved from.
619 address_t address = 0;
620
621 // We found no cache with the required index -> we are going to use the next cache.
622 // NB: we must take care of the actual number of caches.
623 idxCache_ = (idxCache_+1) % parent_t::CACHE_NB; // should be optimized by the compiler
624
625 previousBlockIdx_[idxCache_] = blockIdx;
626
627 address_t beg=0,end=0;
628
629 if (this->hasBeenFilled())
630 {
631 // If the vector was filled (ie. call to 'fill'), we know that the blocks are
632 // contiguous in memory, so we don't need to use the memory tree then.
633 address = this->getFillAddress() + blockIdx*SIZEOF_DATABLOCK;
634
635 beg = this->getFillAddress();
636 end = beg + sizeof(T)*this->size();
637 }
638 else
639 {
640 // from the memory tree, we get the address of the block to be used
641 address = memoryTree_[blockIdx];
642 }
643
644 VERBOSE_VECTOR ("vector::operator[] idxCache_: %ld current 0x%lx and next 0x%lx hasBeenFilled: %ld \n",
645 uint64_t(idxCache_),
646 uint64_t(this->cache(idxCache_).address()),
647 uint64_t(address), uint64_t(this->hasBeenFilled())
648 );
649
650 if (address != this->cache(idxCache_).address())
651 {
652 if (this->cache(idxCache_).address()!=0) { this->cache(idxCache_).update(); }
653
654 // we read the new block from the memory.
655 // We also provide the memory boundaries of the vector
656 this->cache(idxCache_).read (address, beg,end);
657 }
658 }
659 else
660 {
661 // We found a cache with the required index.
662 idxCache_ = look;
663 }
664
665 VERBOSE_VECTOR ("vector::operator[%4ld] block: %ld look: %ld @cache: 0x%08lx address: [0x%08lx] idxCache_: %ld previousBlockIdx_: [%ld] -----------------> changed [%c] val: %ld\n",
666 (uint64_t)idx,
667 (uint64_t)(cache_t::getBlockIdx(idx)),
668 uint64_t(look),
669 (uint64_t)(&(this->cache(idxCache_))),
670 (uint64_t)(this->cache(idxCache_).address()),
671 uint64_t(idxCache_),
672 (uint64_t)previousBlockIdx_[idxCache_],
673 look==CACHE_NB ? 'X' : ' ',
674 (uint64_t)this->cache(idxCache_) [idx]
675 );
676
677 return this->cache(idxCache_) [idx];
678 }
