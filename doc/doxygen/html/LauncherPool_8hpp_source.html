88 template<template<typename ...> class Task, typename...Traits, typename...Args, typename Callback>
89 auto submit (Callback cbk, Args&&...args)
90 {
91 // We need to be sure that the arguments to be used for task execution will have a life cycle long enough.
92 // This is important for instance when we directly use 'split' on client side, which implies that this argument
93 // is deleted as soon as the call to 'submit' is done. So we make sure here to keep the information in a 'Command'
94 // function object that will be used in the 'detach_task' call.
95 struct Command
96 {
97 LauncherPool& parent;
98 Callback cbk;
99 std::tuple<std::decay_t<Args>...> args;
100
101 auto operator() ()
102 {
103 const std::optional<std::size_t> idx = BS::this_thread::get_index();
104
105 if (idx)
106 {
107 [[maybe_unused]]std::size_t currentRequestId = 0; {
108 std::lock_guard<std::mutex> g (parent.mutex_);
109 currentRequestId = parent.requestId_++;
110 }
111
112 // We retrieve the launcher matching the current index
113 launcher_t& launcher = parent.getLauncher(*idx);
114
115 [[maybe_unused]]auto t1 = std::chrono::high_resolution_clock::now();
116
117 auto&& results = std::apply ([&] (auto&&... theargs) {
118 // We launch the task and return its result.
119 return launcher.template run<Task,Traits...> (std::forward<decltype(theargs)>(theargs)...);
120 }, args);
121
122 [[maybe_unused]]auto t2 = std::chrono::high_resolution_clock::now();
123 [[maybe_unused]]auto dt1 = std::chrono::duration_cast<std::chrono::milliseconds>(t1-parent.t0_).count();
124 [[maybe_unused]]auto dt2 = std::chrono::duration_cast<std::chrono::milliseconds>(t2-parent.t0_).count();
125
126 // uncomment this for having "bargraph" stats.
127 parent.stats_[*idx].emplace_back (currentRequestId, dt1, dt2);
128
129 // We call the callback with the results
130 cbk (launcher, std::move(results));
131 }
132 else
133 {
134 throw std::runtime_error ("thread pool failed to process the task");
135 }
136 }
137 };
138 threadpool_->detach_task ( Command { *this, cbk, std::make_tuple(std::forward<Args>(args)...) } );
139 }

155 auto dump() const {
156 const int width = 130;
157 // we look for the max time
158 std::size_t maxtime = 0;
159 for (std::size_t i=0; i<stats_.size(); ++i) {
160 for (std::size_t j=0; j<stats_[i].size(); ++j) {
161 auto const& st = stats_[i][j];
162 if (st.end > maxtime) { maxtime=st.end; }
163 }
164 }
165 float div = (float)width/(float)maxtime;
166 printf ("width: %d maxtime: %ld div: %.lf\n", width, maxtime, div);
167 size_t nbFilled = 0;
168 char buf[width+1] = {'.'};
169 for (std::size_t i=0; i<stats_.size(); ++i) {
170 for (auto& c : buf) { c='.'; }
171 for (std::size_t j=0; j<stats_[i].size(); ++j) {
172 auto const& st = stats_[i][j];
173 for (int k=st.begin*div; k<=st.end*div and k<width; k++) {
174 buf[k]='0' + (j%10);
175 }
176 }
177 buf[width] = 0;
178 for (auto c : buf) { nbFilled += c=='.' ? 1 : 0; }
179 printf ("[%3ld] %s\n", i, buf);
180 }
181 printf ("usage: %.3f\n", 1.0 - (double) nbFilled / (stats_.size()*width));
182 }
