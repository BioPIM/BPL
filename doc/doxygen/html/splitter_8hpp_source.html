17 // Enumeration holding different parallelization schemes.
18 // NOTE: not used right now (only CONT is relevant)
19 enum SplitKind
20 {
21 CONT = 0, // split by contiguous chunks; ex: [0:5] => [0,1,2] [2,3,4]
22 RAKE = 1, // interleave split; ex: [0:5] => [0,2,4] [1,3,5]
23 RAND = 2 // random split ex: [0:5] => [0,3,5] [1,2,4]

253{
254 // 0: no split
255 // 1: split at rank level

258 static_assert (sizeof...(ARGS)<=32);
