159 template<template<typename ...> class TASK, typename...TRAITS, typename ...ARGS>
160 auto run (ARGS&&...args)
161 {
162 using task_t = TASK<arch_t,TRAITS...>;
163
164 // We determine the result type of the TASK::run method.
165 // Note that the TASK type is instantiated with the current ARCH type.
166 using result_t = std::decay_t<bpl::return_t<decltype(&task_t::operator())>>;
167
168 // We create the result vector.
169 std::vector<result_t> results(getProcUnitNumber());
170
171 std::vector<float> exectimes(getProcUnitNumber());
172
173 threadpool_->detach_sequence (0, getProcUnitNumber(), [&] (std::size_t idx)
174 {
175 task_t task;
176
177 // We may have to configure the task, according to the fact that its class inherits (or not) from bpl:Task
178 configure (
179 task,
180 idx, // process unit uid
181 idx, // group uid: same as thread
182 0, // me id
183 0
184 );
185
186 // we check whether an argument has to be split.
187 if constexpr(count_predicate_match_v<hasSplitArgument, ARGS...> == 0)
188 {
189 TimeStamp ts (exectimes[idx]);
190 results[idx] = std::move(task (std::forward<decltype(args)>(args)...));
191 }
192 else
193 {
194 TimeStamp ts (exectimes[idx]);
195
196 // We get the task parameters and remove tags if any
197 using params_t = tuple_projection<
198 remove_all_tags,
199 typename bpl::args<decltype(&task_t::operator())>::as_tuple
200 >::type;
201
202 // We look whether one of the task parameter is a left reference.
203 [[maybe_unused]] constexpr int hasNonConstRefArg2 = bpl::CheckPredicate<
204 bpl::is_noconst_lvalue_reference,
205 params_t
206 >::value;
207
208 constexpr int hasNonConstRefArg = true;
209
210 // We need to associate each argument with its index, since we might have use type
211 // information of the nth parameter of the task (see 'prepare')
212 auto exec = [&] <std::size_t...Is> (std::index_sequence<Is...>, auto&&...args) {
213 // We might need an intermediate tuple here in order to have non temporary arguments to provide to the task.
214 // This is important when a task parameter is a non const reference (eg. sort)
215 if constexpr (hasNonConstRefArg)
216 {
217 auto tup = std::make_tuple (prepare<TASK,Is> (std::forward<decltype(args)>(args), idx, getProcUnitNumber())...);
218 return std::apply ( [&](auto &&... args) { return task (std::forward<decltype(args)>(args)...); },
219 tup
220 );
221 }
222 else {
223 return task(prepare<TASK,Is>(std::forward<decltype(args)>(args), idx, getProcUnitNumber())...);
224 }
225 };
226
227 results[idx] = std::move(exec (
228 std::index_sequence_for<ARGS...>{},
229 std::forward<decltype(args)>(args)...
230 ));
231 }
232 });
233
234 threadpool_->wait();
235
236 auto maxTime = *std::max_element(exectimes.begin(), exectimes.end());
237 statistics_.addTiming("run/once/launch", maxTime);
238
239 // We return the partial results as a vector.
240 return results;
241 }

247 template<template<typename ...> class TASK, std::size_t Idx>
248 decltype(auto) prepare (auto&& x, size_t idx, size_t nbitems)
249 {
250 using task_t = TASK<arch_t>;
251 using dtype = std::decay_t<decltype(x)>;
252
253 if constexpr (impl::GetSplitStatus<dtype,lowest_level_t>::value > 0)
254 {
255 // we retrieve the type of the task parameter for the input argument 'x'.
256 using arg_t = std::decay_t<bpl::parameter_t<decltype(&task_t::operator()),Idx>>;
257
258 // We remove potential tags from the type.
259 using notags_t = std::decay_t<std::tuple_element_t<0, bpl::transform_tuple_t <
260 std::tuple<arg_t>,
261 bpl::removetag_once,
262 bpl::removetag_global
263 >>>;
264
265 // We have to consider the type of the corresponding parameter of the task.
266 // If it is a span, we call split_view, otherwise we call split
267 if constexpr(bpl::is_span<notags_t>::value) {
268 return SplitOperator<dtype>::split_view (std::forward<decltype(x)>(x), idx,nbitems);
269 }
270 else {
271 return SplitOperator<dtype>::split (std::forward<decltype(x)>(x), idx,getProcUnitNumber());
272 }
273 }
274 else
275 {
276 return std::forward<decltype(x)>(x);
277 }
278 }
