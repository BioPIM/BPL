59struct TimeStats : TimeStatsValues<uint64_t>
60{
61 using value_type = uint64_t;
62
63 static auto minmax (const std::vector<TimeStats>& entries)
64 {
65 TimeStats m;
66 TimeStats M;
67
68 m.all = std::numeric_limits<value_type>::max();
69 M.all = 0;
70
71 std::vector<value_type> alls;
72 for (const auto& x : entries)
73 {
74 alls.push_back (x.all);
75 if (x.all < m.all) { m = x; }
76 if (x.all > M.all) { M = x; }
77 }
78
79#ifdef WITH_STATS_TIME_MEDIAN
80 std::sort(alls.begin(), alls.end());
81#endif
82 std::array<value_type,32> quantiles = {};
83
84 if (not alls.empty())
85 {
86 size_t idx=0; for (auto& x : quantiles) { x = alls[idx*alls.size()/quantiles.size()]; idx++; }
87 quantiles.back() = alls.back();
88 }
89
90 return std::make_tuple (m,M,quantiles);
91 }
92
93 static auto mean (const std::vector<TimeStats>& entries)
94 {
95 TimeStatsValues<uint64_t> res;
96
97 for (const auto& x : entries) { res += x; }
98
99 return TimeStatsValues<double>
100 {
101 double(res.unserialize) / double(res.all),
102 double(res.split) / double(res.all),
103 double(res.exec) / double(res.all),
104 double(res.result) / double(res.all),
105 double(res.all) / double(entries.size())
106 };
107 }
