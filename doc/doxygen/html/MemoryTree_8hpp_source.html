146 auto insert (address_t a)
147 {
148 DEBUG_MEMTREE("[MemoryTree::insert] this: 0x%lx a=0x%lx bounds: [0x%lx,0x%lx] \n",
149 uint64_t(this),
150 uint64_t(a),
151 uint64_t(leavesBounds_.first),
152 uint64_t(leavesBounds_.second)
153 );
154
155 // we increase the number of inserted leaves.
156 nbLeaves_ ++;
157
158 previousBlockIdx_ = nbLeaves_ >> NBITEMS_PER_BLOCK_LOG2;
159
160 // we insert the item into the stack.
161 stack_.insert (a);
162
163 // we increase the number of items for depth 0 (ie. leaves)
164 countsPerDepth_[LEAF_LEVEL] ++;
165
166 // We may have to flush the content of the stack. We begin with leaf level and go upwards in the tree.
167 // Each full block will be merged, ie. copied in memory at a given address that will be registered
168 // in the parent block.
169 flush ([&](depth_t depth) { return countsPerDepth_[depth] == NBITEMS_PER_BLOCK; });
170
171 // we set the tree as "dirty" -> other methods could need to do a 'flush'
172 isDirty_ = true;
173
174 return a;
175 }

197 address_t operator[] (size_t idx)
198 {
199 address_t result {};
200
201 DEBUG_MEMTREE ("\n----------------------------------------------------------------------------------------\n");
202
203 DEBUG_MEMTREE ("Memtree operator[] idx: %ld BlockIdx: [%ld,%ld] StackLeavesNb: %ld case: %ld\n",
204 (uint64_t)idx,
205 (uint64_t)idx>>NBITEMS_PER_BLOCK_LOG2,
206 (uint64_t)previousBlockIdx_,
207 (uint64_t)getStackLeavesNb(),
208 (uint64_t) (idx>>NBITEMS_PER_BLOCK_LOG2 == size()>>NBITEMS_PER_BLOCK_LOG2)
209 );
210 debug("DEBUG FOR operator[]");
211
212 address_t* tail = stack_.tail();
213 int notaligned = uint64_t(tail) % 8 != 0;
214
215 if (notaligned) { tail = (address_t*) (uint64_t(tail+7) & ~7); }
216
217 // The required block may be already there (in the cache)
218 if (idx>>NBITEMS_PER_BLOCK_LOG2 == size()>>NBITEMS_PER_BLOCK_LOG2 )
219 {
220 result = getStackLeaves() [idx & NBITEMS_PER_BLOCK_MASK];
221 }
222
223 // The required block may be already there (after the cache)
224 else if (idx>>NBITEMS_PER_BLOCK_LOG2 == previousBlockIdx_)
225 {
226 result = tail [idx & NBITEMS_PER_BLOCK_MASK];
227 }
228 else
229 {
231 normalize();
232
233 size_t requiredBlockIdx = 0;
234 size_t requiredDepth = 0;
235 iterate_stack ([&](size_t depth, address_t value)
236 {
237 size_t p=1;
238 for (size_t d=1; d<=depth; d++) { p *= NBITEMS_PER_BLOCK; }
239 requiredBlockIdx += p;
240 requiredDepth = depth;
241 result = value;
242 return idx >= requiredBlockIdx;
243 });
244
245 // accessing the required leaf consists in traversing the tree like we enumerate the digits of a number in a given basis.
246 for (depth_t power=requiredDepth-1; power>=0; power--)
247 {
248 size_t digit = (idx >> (power*NBITEMS_PER_BLOCK_LOG2)) & NBITEMS_PER_BLOCK_MASK;
249
250 // we read a block from memory and put it at the end of the stack.
251 // note: we use the inner buffer of the stack here, ie. we don't use the 'insert' method
252 // -> we can do that because we have had a supplementary level during the stack declaration
253 // which allows to address NBITEMS_PER_BLOCK items
254 // Note also that we want to go directly to a specific leaf, so we don't need to keep information
255 // of intermediate nodes.
256 address_t* ptr = ALLOCATOR::read ((void*)result , tail, sizeof(result)*(NBITEMS_PER_BLOCK));
257
258 DEBUG_MEMTREE ("Memtree operator[] idx=%ld power=%ld digit=%ld #stack=%ld result: 0x%lx ptr: 0x%lx tail: 0x%lx notaligned: %ld\n",
259 (uint64_t)idx,
260 (uint64_t)power,
261 (uint64_t)digit,
262 (uint64_t)stack_.idx_,
263 (uint64_t)result,
264 (uint64_t)ptr,
265 (uint64_t)tail,
266 (uint64_t)notaligned
267 );
268
269 // we update the result.
270 result = ptr[digit /* + (notaligned!=0 ? 1 : 0)*/];
271 }
272
273 // we remember the required idx for potentially optimize lookup for next call to operator[]
274 previousBlockIdx_ = idx >> NBITEMS_PER_BLOCK_LOG2;
275
276 DEBUG_MEMTREE ("Memtree operator[] previousBlockIdx_: %ld result: 0x%lx \n",
277 (uint64_t)previousBlockIdx_,
278 (uint64_t)result
279 );
280 }
281
282 return result;
283 }

304 template<typename FCT>
305 auto iterate_stack (FCT fct)
306 {
307 bool result = true;
308
309 if (stack_.idx_==0) { return result; } // needed if tree is empty
310
311 depth_t depth = maxDepth_;
312
313 size_t i=0;
314 for (size_t count = countsPerDepth_ [depth]; depth >= 0; ++i)
315 {
316 // we notify the depth and the item of the stack.
317 result = fct (depth, stack_.items_[i]);
318 if (result==false) { break; }
319
320 if (--count==0) // go to next depth
321 {
322 for (--depth; depth>=0; --depth)
323 {
324 // we set the new count for the new level.
325 count = countsPerDepth_ [depth];
326
327 // in case the new level has some items, we must break and not let the --depth of the for loop
328 if (count>0) { break; }
329 }
330 }
331 }
332 return result;
333 }

422 auto merge (depth_t depth, bool IsLeaf)
423 {
424 constexpr int32_t NbItemsToWrite = std::max(LEAVES_NBITEMS_PER_BLOCK, NBITEMS_PER_BLOCK);
425
426 size_t nbItems = countsPerDepth_[depth];
427
428 // we add empty values to the end of the stack in order to have a complete block
429 for (size_t i=nbItems ; i < NbItemsToWrite; i++) { stack_.insert (address_t{}); }
430
431 // At leaf level, we update the previous link
432 if (IsLeaf)
433 {
434 if (leavesBounds_.second != 0) { stack_.get(-NbItemsToWrite)[IDX_BLOCK_PREVIOUS] = leavesBounds_.second; }
435 }
436
437 address_t* from = stack_.get(-NbItemsToWrite);
438
439 address_t b = 0;
440
441
442 if (uint64_t(from)%8 == 0)
443 {
444 b = ALLOCATOR::write (from, sizeof(address_t) * NbItemsToWrite);
445 }
446 else
447 {
448 // Enforce alignment
449 address_t tmp [NbItemsToWrite];
450 for (int i=0; i<NbItemsToWrite; i++) { tmp[i] = from[i]; }
451
452 b = ALLOCATOR::write (tmp, sizeof(address_t) * NbItemsToWrite);
453 }
454
455 DEBUG_MEMTREE ("[MemoryTree::merge] from: 0x%lx [%c] bounds: [0x%lx,0x%lx] #items: %ld depth: %ld block of %2ld written at 0x%lx -> ",
456 uint64_t(from),
457 uint64_t(from)%8==0 ? ' ' : '!',
458 uint64_t(leavesBounds_.first),
459 uint64_t(leavesBounds_.second),
460 uint64_t(nbItems),
461 uint64_t(depth),
462 uint64_t(NbItemsToWrite),
463 uint64_t(b)
464 );
465 for (size_t i=0; i<NbItemsToWrite; i++) { DEBUG_MEMTREE ("0x%lx ", uint64_t(stack_.get(-NbItemsToWrite)[i])); }
466 DEBUG_MEMTREE ("\n");
467
468 // At leaf level, we update the next link
469 if (IsLeaf)
470 {
471 if (leavesBounds_.second != 0)
472 {
473 // we update the next link of the previous leaf
474 address_t* ptr = (address_t*)leavesBounds_.second + IDX_BLOCK_NEXT;
475
476 ALLOCATOR::writeAtomic ((address_t)ptr, b);
477
478 DEBUG_MEMTREE ("[MemoryTree::merge] MEMORY UPDATE FOR NEXT LINK: 0x%lx (0x%lx) -> 0x%lx IDX_BLOCK_NEXT: %ld \n",
479 uint64_t(leavesBounds_.second),
480 uint64_t(ptr),
481 uint64_t(b),
482 uint64_t(IDX_BLOCK_NEXT)
483 );
484 }
485
486 // We remember the last seen leaf.
487 leavesBounds_.second = b;
488 DEBUG_MEMTREE ("[MemoryTree::merge] SETTING LAST : 0x%lx\n", uint64_t(b));
489
490 // We remember the first leaf.
491 if (leavesBounds_.first==0)
492 {
493 leavesBounds_.first = b;
494 DEBUG_MEMTREE ("[MemoryTree::merge] SETTING FIRST: 0x%lx\n", uint64_t(b));
495 }
496 }
497
498 // we pop this block of addresses from the stack
499 stack_.pop (NbItemsToWrite);
500
501 // we insert the new block.
502 stack_.insert (b);
503
504 // we update the counts for the current level (goes back to 0) and increase the next one.
505 countsPerDepth_[depth+0] = 0;
506 countsPerDepth_[depth+1] += 1;
507
508 if (maxDepth_ < depth+1) { maxDepth_ = depth+1; }
509 }

536 template<bool FORWARD>
537 struct leaves_iterator_generic
538 {
539 using tree_t = MemoryTree<ALLOCATOR,NBITEMS_PER_BLOCK_LOG2,MAX_MEMORY_LOG2>;
540
541 // We may have to use the content of the stack
542 address_t* stackLeaves = nullptr;
543
544 // Number of leaves present in the stack.
545 size_t nbStackLeaves = 0;
546
547 address_t* leavesStorage = nullptr;
548
549 // Pointer holding the information to be iterated; it may point to the stack or to the storage
550 // specific to the iterator.
551 address_t* storage = nullptr;
552
553 // We need to know what is the next block of information to be used.
554 address_t next = 0;
555
556 // i/end are the indexes bounds to be used for iterating 'storage'
557 ssize_t i = 0;
558 ssize_t end = 0;
559
560 // Number of call to operator++. Will be used for operator!=
561 size_t n = 0;
562 size_t nbleaves = 0;
563
564 const size_t link = FORWARD ? IDX_BLOCK_NEXT : IDX_BLOCK_PREVIOUS;
565 const int direction = FORWARD ? 1 : -1;
566
567 address_t* round (address_t* a) { return (address_t*) ((uint64_t(a) + 7) & ~7); }
568
569 leaves_iterator_generic (tree_t& ref) : nbleaves(ref.size())
570 {
571 stackLeaves = ref.stack_.tail() - ref.countsPerDepth_[LEAF_LEVEL];
572 leavesStorage = round (ref.stack_.tail());
573
574 // We remember the number of leaves stored in the stack.
575 nbStackLeaves = ref.countsPerDepth_[LEAF_LEVEL];
576
577 // According to FORWARD, we get the head or the tail of the double linked list encoded with the leaves of the tree
578 address_t bound = FORWARD ? ref.leavesBounds_.first : ref.leavesBounds_.second;
579
580 DEBUG_MEMTREE ("[MemoryTree::iterator::iterator] @ref: 0x%lx bounds: [0x%lx,0x%lx] it: 0x%lx i: %ld nbleaves: %ld LEAVES_NBITEMS_PER_BLOCK: %ld nbStackLeaves: %ld stackLeaves: 0x%lx leavesStorage: 0x%lx #stack: %ld\n",
581 uint64_t(&ref),
582 (uint64_t)ref.leavesBounds_.first,
583 (uint64_t)ref.leavesBounds_.second,
584 (uint64_t)bound,
585 (uint64_t)i,
586 (uint64_t)nbleaves,
587 (uint64_t)LEAVES_NBITEMS_PER_BLOCK,
588 (uint64_t)nbStackLeaves,
589 (uint64_t)stackLeaves,
590 (uint64_t)leavesStorage,
591 (uint64_t)ref.stack_.size()
592 );
593
594 if (nbleaves>0) { configure <true> (bound); }
595 }

616 template<bool FIRST>
617 void configure_forward (address_t block)
618 {
619 if (block!=0)
620 {
621 storage = leavesStorage;
622
623 ALLOCATOR::read ((void*)block, storage, LEAVES_NBITEMS_PER_BLOCK*sizeof(address_t));
624
625 i = 0;
626 end = NBITEMS_PER_BLOCK;
627 next = storage[link]!=0 ? storage[link] : 0;
628
629 DEBUG_MEMTREE ("[MemoryTree::configure_forward] block: 0x%lx storage: 0x%lx size: %ld [i,end]: %ld %ld \n",
630 uint64_t((void*)block),
631 uint64_t(storage),
632 uint64_t(LEAVES_NBITEMS_PER_BLOCK*sizeof(address_t)),
633 uint64_t(i),
634 uint64_t(end)
635 );
636 }
637 else
638 {
639 storage = stackLeaves;
640
641 DEBUG_MEMTREE ("[MemoryTree::configure_forward] DUMP STACK: \n");
642
643 i = 0;
644 end = nbleaves % NBITEMS_PER_BLOCK;
645 next = 0;
646 }
647 }

649 template<bool FIRST>
650 void configure_backward (address_t block)
651 {
652 if constexpr (FIRST)
653 {
654 // The storage is the current leaves stack.
655 storage = stackLeaves;
656
657 if (nbStackLeaves==0) { ALLOCATOR::read ((void*)block, storage, LEAVES_NBITEMS_PER_BLOCK*sizeof(address_t)); }
658
659 // We compute the index bounds
660 i = (nbleaves-1) % NBITEMS_PER_BLOCK;
661 end = -1;
662
663 // Next storage.
664 // Note: if nbStackLeaves==0, it means that the previous block has been flushed
665 // BUT the stack still holds the leaves => we can then avoid to read again this block.
666 next = nbStackLeaves==0 ? storage[link] : block;
667 }
668 else if (next != 0)
669 {
670 // The storage is beyond the current leaves stack.
671 storage = leavesStorage;
672
673 ALLOCATOR::read ((void*)block, storage, LEAVES_NBITEMS_PER_BLOCK*sizeof(address_t));
674
675 // We compute the index bounds
676 i = NBITEMS_PER_BLOCK-1;
677 end = -1;
678
679 // Next storage block (previous or next block in the double linked list).
680 next = storage[link];
681 }
682 }

782 template<bool REVERSED, typename FCT> void DFS_iter (FCT fct, depth_t depth, address_t node)
783 {
784 auto swap = [] (address_t& a, address_t& b) { address_t c(a); a=b; b=c; };
785
786 auto set_depth = [] (address_t node, depth_t d) { return node | address_t(d) << 8*(sizeof(address_t)-sizeof(depth_t)); };
787 auto get_depth = [] (address_t node) { return depth_t(node >> 8*(sizeof(address_t)-sizeof(depth_t))); };
788 auto get_address = [] (address_t node) { return node & ((address_t(1) << 8*(sizeof(address_t)-sizeof(depth_t)))-1); };
789
790 size_t keepSize = stack_.size();
791
792 // WARNING !!! in order to keep low stack memory usage (important for UPMEM arch),
793 // we "patch" the address value by adding the depth as the most significant byte
794 // -> it makes it possible to have only one stack and not a dedicated stack for storing the node depths.
795 stack_.insert (set_depth(node,depth));
796
797 // We will use a variable that keep track of the previous node depth
798 // (remember that depth=0 stands for the leaves and intermediate nodes have higher depth).
799 // So when the current node depth is equal to prevDepth+1, it means that we currently traverse
800 // a node whose children have just been traversed (ie. with a depth equal to prevDepth)
801 uint8_t prevDepth = depth;
802
803 // Here is our iteration. Note that we should begin the iteration with the root node present twice,
804 // we iterate as long as there is only one node left in the stack (the root node).
805 while (stack_.size() > keepSize)
806 {
807 // We get the top of the stack and remove it from the stack.
808 address_t head = stack_.top();
809
810 // We get the actual address and depth from it.
811 node = get_address(head);
812
813 depth_t d = get_depth(head);
814
815 // We check whether we have finished all children.
816 bool areChildrenDone = d == (prevDepth+1);
817
818 if (node==address_t())
819 {
820 stack_.pop();
821 continue;
822 }
823
824 // We have here a non-leaf node that has not yet traversed its children.
825 if (d > 0 and not areChildrenDone)
826 {
827 // We insert all the children into the stack.
828 address_t* block = ALLOCATOR::read ((void*)node, stack_.tail(), NBITEMS_PER_BLOCK*sizeof(address_t));
829 stack_.shift (NBITEMS_PER_BLOCK);
830
831 // And into the depth stack as well with a lesser depth 'd-1' than the current node.
832 for (size_t k=0; k<NBITEMS_PER_BLOCK; k++) { block[k] = set_depth (block[k], d-1); }
833
834 // If we want insertion order, we have to revert the order of the retrieved data.
835 // If we want reverse order, we don't have to do anything.
836 // NOTE: NBITEMS_PER_BLOCK must be power of 2 to make 'swap' efficient.
837
838 if (not REVERSED)
839 {
840 for (size_t k=0; k<NBITEMS_PER_BLOCK/2; k++) { swap (block[k], block[NBITEMS_PER_BLOCK-k-1]); }
841 }
842 }
843
844 // Otherwise, it means that we have:
845 // - a leaf node
846 // - or a non-leaf node with all its children just traversed
847 // We just traverse this node and remove it from the stack.
848 else
849 {
850 // We call the functor with the retrieved value and its associated depth.
851 fct (d, node);
852
853 // We memorize the current depth for the next iteration.
854 prevDepth = d;
855
856 // This node has been processed, we remove it from the stack.
857 stack_.pop();
858 }
859 }
860 }
